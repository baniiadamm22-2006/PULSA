public static function form(\Filament\Forms\Form $form): \Filament\Forms\Form
{
    return $form->schema([]);
}

public static function infolist(\Filament\Infolists\Infolist $infolist): \Filament\Infolists\Infolist
{
    return $infolist->schema([
            \Filament\Infolists\Components\TextEntry::make('order.invoice_number')->label('Invoice'),
            \Filament\Infolists\Components\TextEntry::make('product_name_snapshot')->label('Nama Produk Saat Dipesan'),
            \Filament\Infolists\Components\TextEntry::make('product_code_snapshot')->label('Kode Produk'),
            \Filament\Infolists\Components\TextEntry::make('quantity')->label('Jumlah'),
            \Filament\Infolists\Components\TextEntry::make('cost_price_snapshot')->label('Harga Modal Saat Dipesan')->money('IDR'),
            \Filament\Infolists\Components\TextEntry::make('selling_price_snapshot')->label('Harga Jual Saat Dipesan')->money('IDR'),
            \Filament\Infolists\Components\TextEntry::make('total_amount')->label('Total')->money('IDR'),
            \Filament\Infolists\Components\TextEntry::make('created_at')->label('Dibuat')->dateTime('d M Y H:i')
    ])->columns(2);
}

public static function table(\Filament\Tables\Table $table): \Filament\Tables\Table
{
    return $table->columns([
            \Filament\Tables\Columns\TextColumn::make('order.invoice_number')->label('Invoice')->searchable()->sortable(),
            \Filament\Tables\Columns\TextColumn::make('product_name_snapshot')->label('Produk')->searchable()->sortable(),
            \Filament\Tables\Columns\TextColumn::make('product_code_snapshot')->label('SKU'),
            \Filament\Tables\Columns\TextColumn::make('quantity')->label('Jumlah')->numeric()->sortable(),
            \Filament\Tables\Columns\TextColumn::make('total_amount')->label('Total')->money('IDR')->sortable(),
            \Filament\Tables\Columns\TextColumn::make('created_at')->label('Dibuat')->dateTime('d M Y H:i')->sortable()
    ])
        ->actions([\Filament\Tables\Actions\ViewAction::make()])
        ->defaultSort('id', 'desc')
        ->emptyStateHeading('Belum ada data');
}
