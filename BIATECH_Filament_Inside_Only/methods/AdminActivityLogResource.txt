public static function form(\Filament\Forms\Form $form): \Filament\Forms\Form
{
    return $form->schema([]);
}

public static function infolist(\Filament\Infolists\Infolist $infolist): \Filament\Infolists\Infolist
{
    return $infolist->schema([
            \Filament\Infolists\Components\TextEntry::make('user.email')->label('Admin'),
            \Filament\Infolists\Components\TextEntry::make('action')->label('Aksi'),
            \Filament\Infolists\Components\TextEntry::make('subject_type')->label('Tipe Objek'),
            \Filament\Infolists\Components\TextEntry::make('subject_id')->label('ID Objek'),
            \Filament\Infolists\Components\TextEntry::make('created_at')->label('Waktu')->dateTime('d M Y H:i')
    ])->columns(2);
}

public static function table(\Filament\Tables\Table $table): \Filament\Tables\Table
{
    return $table->columns([
            \Filament\Tables\Columns\TextColumn::make('user.email')->label('Admin'),
            \Filament\Tables\Columns\TextColumn::make('action')->label('Aksi')->searchable()->sortable(),
            \Filament\Tables\Columns\TextColumn::make('subject_type')->label('Objek')->limit(55),
            \Filament\Tables\Columns\TextColumn::make('subject_id')->label('ID Objek')->numeric()->sortable(),
            \Filament\Tables\Columns\TextColumn::make('created_at')->label('Waktu')->dateTime('d M Y H:i')->sortable()
    ])
        ->actions([\Filament\Tables\Actions\ViewAction::make()])
        ->defaultSort('id', 'desc')
        ->emptyStateHeading('Belum ada data');
}
