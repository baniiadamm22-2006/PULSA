public static function form(\Filament\Forms\Form $form): \Filament\Forms\Form
{
    return $form->schema([
        \Filament\Forms\Components\Section::make('Data Supplier')
            ->description('Daftarkan penyedia PPOB. Kode identitas supplier harus unik.')
            ->schema([
                \Filament\Forms\Components\TextInput::make('name')
                    ->label('Nama Supplier')->required()->maxLength(160),
                \Filament\Forms\Components\TextInput::make('code')
                    ->label('Kode Supplier')->required()->maxLength(64)
                    ->unique(ignoreRecord: true)
                    ->rule('regex:/^[a-zA-Z0-9_-]+$/')
                    ->helperText('Contoh: DIGIFLAZZ'),
                \Filament\Forms\Components\TextInput::make('configuration_reference')
                    ->label('Referensi konfigurasi')->maxLength(255)
                    ->helperText('Hanya nama konfigurasi, jangan masukkan API key / secret.'),
                \Filament\Forms\Components\Toggle::make('is_active')
                    ->label('Aktif')->default(false)
                    ->helperText('Aktifkan setelah koneksi supplier benar-benar siap.'),
            ])->columns(2),
    ]);
}

public static function table(\Filament\Tables\Table $table): \Filament\Tables\Table
{
    return $table->columns([
        \Filament\Tables\Columns\TextColumn::make('name')->label('Supplier')->searchable()->sortable(),
        \Filament\Tables\Columns\TextColumn::make('code')->label('Kode')->searchable()->copyable(),
        \Filament\Tables\Columns\IconColumn::make('is_active')->label('Aktif')->boolean(),
        \Filament\Tables\Columns\TextColumn::make('updated_at')->label('Diperbarui')->dateTime('d M Y H:i')->sortable(),
    ])->filters([
        \Filament\Tables\Filters\TernaryFilter::make('is_active')->label('Status aktif'),
    ])->actions([
        \Filament\Tables\Actions\EditAction::make(),
    ])->defaultSort('id', 'desc')->emptyStateHeading('Belum ada supplier');
}
